import { useEffect, useState } from 'react';
import { StageConfig, getRandomTip } from '@/lib/loading-config';

interface LoadingProgressProps {
  stages: StageConfig[];
  currentStageIndex: number;
  showTips?: boolean;
}

export function LoadingProgress({ stages, currentStageIndex, showTips = true }: LoadingProgressProps) {
  const [tip, setTip] = useState(getRandomTip());
  const currentStage = stages[currentStageIndex] || stages[0];

  // Rotate tips every 5 seconds
  useEffect(() => {
    if (!showTips) return;
    const interval = setInterval(() => {
      setTip(getRandomTip());
    }, 5000);
    return () => clearInterval(interval);
  }, [showTips]);

  return (
    <div
      className="flex flex-col items-center justify-center py-12 px-8"
      style={{ animation: 'fadeIn 0.3s ease-out' }}
    >
      {/* Success checkmark for completed stages */}
      <div className="flex items-center gap-3 mb-6">
        {currentStageIndex > 0 && (
          <div className="flex items-center gap-1 text-[var(--status-green)] text-sm font-medium">
            <svg className="w-4 h-4" fill="currentColor" viewBox="0 0 20 20">
              <path fillRule="evenodd" d="M16.707 5.293a1 1 0 010 1.414l-8 8a1 1 0 01-1.414 0l-4-4a1 1 0 011.414-1.414L8 12.586l7.293-7.293a1 1 0 011.414 0z" clipRule="evenodd" />
            </svg>
            <span>Brief uploaded</span>
          </div>
        )}
      </div>

      {/* Progress bar */}
      <div className="w-full max-w-md mb-6">
        <div className="flex justify-between text-xs text-[var(--text-muted)] mb-2">
          <span>{currentStage.message}</span>
          <span>{currentStage.percent}%</span>
        </div>
        <div className="w-full h-2 bg-[var(--bg-tertiary)] rounded-full overflow-hidden">
          <div
            className="h-full bg-gradient-to-r from-[var(--expedia-navy)] to-[var(--expedia-yellow)] rounded-full transition-all duration-500 ease-out"
            style={{ width: `${currentStage.percent}%` }}
          />
        </div>
      </div>

      {/* Current action */}
      <p className="text-lg font-medium text-[var(--text-primary)] text-center mb-2">
        {currentStage.message}
      </p>
      {currentStage.subMessage && (
        <p className="text-sm text-[var(--text-muted)] text-center">
          {currentStage.subMessage}
        </p>
      )}

      {/* Animated dots */}
      <div className="flex gap-1.5 mt-4">
        {[0, 1, 2].map((i) => (
          <div
            key={i}
            className="w-2 h-2 rounded-full bg-[var(--expedia-navy)]"

              animation: 'dotPulse 1.4s ease-in-out infinite',
              animationDelay: `${i * 160}ms`,

          />
        ))}
      </div>

      {/* Tips section */}
      {showTips && (
        <div
          className="mt-8 p-4 rounded-xl bg-[var(--bg-secondary)] border border-[var(--border-color)] max-w-sm"

          </div>
        </div>
      )}

      <style jsx>{`
        @keyframes fadeIn {
          from { opacity: 0; transform: translateY(8px); }
          to { opacity: 1; transform: translateY(0); }
        }
        @keyframes dotPulse {
          0%, 80%, 100% { transform: scale(0.8); opacity: 0.5; }
          40% { transform: scale(1.2); opacity: 1; }
        }
      `}</style>

  );

```typescript
import { useState, useCallback, useRef } from 'react';
import { StageConfig } from '@/lib/loading-config';

export function useLoadingProgress(stages: StageConfig[]) {
  const [currentStageIndex, setCurrentStageIndex] = useState(0);
  const [isActive, setIsActive] = useState(false);
  const timeoutRef = useRef<NodeJS.Timeout | null>(null);

  const start = useCallback(() => {
    setIsActive(true);
    setCurrentStageIndex(0);
  }, []);

  const advanceStage = useCallback(() => {
    setCurrentStageIndex((prev) => {
      const next = prev + 1;
      if (next >= stages.length) {
        return prev;
      }
      return next;
    });
  }, [stages.length]);

  const complete = useCallback(() => {
    // Jump to complete stage
    setCurrentStageIndex(stages.length - 1);

    // Clear after brief delay
    if (timeoutRef.current) clearTimeout(timeoutRef.current);
    timeoutRef.current = setTimeout(() => {
      setIsActive(false);
      setCurrentStageIndex(0);
    }, 500);
  }, [stages.length]);

  const reset = useCallback(() => {
    if (timeoutRef.current) clearTimeout(timeoutRef.current);
    setIsActive(false);
    setCurrentStageIndex(0);
  }, []);

  // Auto-advance through stages with minimum durations
  const runSimulatedProgress = useCallback(async () => {
    start();

    for (let i = 0; i < stages.length - 1; i++) {
      await new Promise((resolve) => setTimeout(resolve, stages[i].minDuration));
      setCurrentStageIndex(i + 1);
    }
  }, [stages, start]);

  return {
    currentStageIndex,
    currentStage: stages[currentStageIndex],
    isActive,
    start,
    advanceStage,
    complete,
    reset,
    runSimulatedProgress,
  };
}
```

```typescript
'use client';

import { useState } from 'react';
import { BudgetDetails } from '@/lib/types';

interface ProductionBudgetProps {
  extractedTotal?: string;
  onConfirm: (budget: BudgetDetails) => void;
  onSkip?: () => void;
  initialValue?: BudgetDetails | null;
}

const CURRENCIES = [
  { code: 'USD', symbol: '$', label: 'US Dollar' },
  { code: 'GBP', symbol: '£', label: 'British Pound' },
  { code: 'EUR', symbol: '€', label: 'Euro' },
  { code: 'AUD', symbol: 'A$', label: 'Australian Dollar' },
];

export function ProductionBudget({ extractedTotal, onConfirm, onSkip, initialValue }: ProductionBudgetProps) {
  const [totalBudget, setTotalBudget] = useState(initialValue?.totalBudget || extractedTotal || '');
  const [productionBudget, setProductionBudget] = useState(initialValue?.productionBudget || '');
  const [currency, setCurrency] = useState(initialValue?.currency || 'USD');
  const [showHelp, setShowHelp] = useState(false);

  const selectedCurrency = CURRENCIES.find((c) => c.code === currency) || CURRENCIES[0];
  const isValid = totalBudget.trim() !== '' && productionBudget.trim() !== '';

  const handleConfirm = () => {
    if (!isValid) return;
    onConfirm({
      totalBudget,
      productionBudget,
      currency,
    });
  };

  // Calculate suggested production budget (10% of total)
  const suggestedProduction = totalBudget
    ? Math.round(parseFloat(totalBudget.replace(/[^0-9.]/g, '')) * 0.1).toLocaleString()
    : null;

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="pb-4 border-b border-[var(--border-color)]">
        <h3 className="text-lg font-semibold text-[var(--text-primary)] mb-1">
          Production Budget
        </h3>
        <p className="text-sm text-[var(--text-muted)]">
          This is what the creative team actually has to work with.
        </p>
      </div>

      {/* Currency selector */}
      <div>
        <label className="block text-sm font-medium text-[var(--text-secondary)] mb-2">
          Currency
        </label>
        <div className="flex gap-2">
          {CURRENCIES.map((c) => (
            <button
              key={c.code}
              onClick={() => setCurrency(c.code)}
              className={`px-4 py-2 rounded-lg text-sm font-medium transition-colors ${
                currency === c.code
                  ? 'bg-[var(--expedia-navy)] text-white'
                  : 'bg-[var(--bg-secondary)] text-[var(--text-secondary)] hover:bg-[var(--bg-tertiary)]'
              }`}
            >
              {c.symbol} {c.code}
            </button>
          ))}
        </div>
      </div>

      {/* Total campaign budget */}
      <div>
        <label className="block text-sm font-medium text-[var(--text-secondary)] mb-2">
          Total Campaign Budget
        </label>
        <div className="relative">
          <span className="absolute left-4 top-1/2 -translate-y-1/2 text-[var(--text-muted)]">
            {selectedCurrency.symbol}
          </span>
          <input
            type="text"
            value={totalBudget}
            onChange={(e) => setTotalBudget(e.target.value)}
            placeholder="500,000"
            className="w-full pl-10 pr-4 py-3 rounded-xl border border-[var(--border-color)] focus:outline-none focus:ring-2 focus:ring-[var(--expedia-navy)]/20 focus:border-[var(--expedia-navy)]"
          />
        </div>
        {extractedTotal && extractedTotal !== totalBudget && (
          <p className="text-xs text-[var(--text-muted)] mt-1">
            Extracted from brief: {extractedTotal}
          </p>
        )}
      </div>

      {/* Production budget - MANDATORY */}
      <div>
        <div className="flex items-center justify-between mb-2">
          <label className="block text-sm font-medium text-[var(--text-secondary)]">
            Production Budget <span className="text-[var(--status-red)]">*</span>
          </label>
          <button
            onClick={() => setShowHelp(!showHelp)}
            className="text-xs text-[var(--expedia-navy)] hover:underline"
          >
            What's this?
          </button>
        </div>

        {showHelp && (
          <div className="mb-3 p-3 rounded-lg bg-[var(--expedia-navy)]/5 border border-[var(--expedia-navy)]/20 text-sm text-[var(--text-secondary)]">
            <p className="mb-2">
              <strong>Production budget</strong> is what the creative team has to make the work —
              video production, photography, design, etc.
            </p>
            <p>
              It's typically around <strong>10% of total campaign budget</strong>, but varies by project.
            </p>
          </div>
        )}

        <div className="relative">
          <span className="absolute left-4 top-1/2 -translate-y-1/2 text-[var(--text-muted)]">
            {selectedCurrency.symbol}
          </span>
          <input
            type="text"
            value={productionBudget}
            onChange={(e) => setProductionBudget(e.target.value)}
            placeholder="50,000"
            className="w-full pl-10 pr-4 py-3 rounded-xl border border-[var(--border-color)] focus:outline-none focus:ring-2 focus:ring-[var(--expedia-navy)]/20 focus:border-[var(--expedia-navy)]"
          />
        </div>

        {suggestedProduction && !productionBudget && (
          <button
            onClick={() => setProductionBudget(suggestedProduction)}
            className="mt-2 text-xs text-[var(--expedia-navy)] hover:underline"
          >
            Use suggested: {selectedCurrency.symbol}{suggestedProduction} (10% of total)
          </button>
        )}
      </div>

      {/* Actions */}
      <div className="flex gap-3 pt-4">
        {onSkip && (
          <button onClick={onSkip} className="btn-outline flex-1">
            Skip for now
          </button>
        )}
        <button
          onClick={handleConfirm}
          disabled={!isValid}
          className="btn-secondary flex-1 flex items-center justify-center gap-2"
        >
          Confirm Budget
          <span>→</span>
        </button>
      </div>

      {!isValid && (
        <p className="text-xs text-[var(--status-amber)] text-center">
          Production budget is required to continue
        </p>
      )}
    </div>
  );
}
```
